   function insertWorks(informations)
{
  $('#imagesAndContent').empty();

  informations.forEach(function(information) {
    let imageURL = "baseImages/"+information.location;
    let title = information.name;
    let description = information.firstDescription;

    let card = $('<div>').addClass('card').attr('alt', name).css({'border' : '0px', 'text-align' : 'center'});

    let imageContainer = $('<div>').addClass('img-container');
    let image = $('<img>').attr('src', imageURL).attr('alt', 'Image');
    image.appendTo(imageContainer);

    // let image = $('<img>').attr('src', imageURL);
    let insertTitle = $('<p>').text(title);
    let formattedDescription = description.replace(/\n/g, '<br>');
    let insertDescription = $('<p>').html(formattedDescription);
    
    card.append(imageContainer, insertTitle, insertDescription);

    card.click(function()
    {
      makeBigPic(imageURL);
    });

    $('#imagesAndContent').append(card);
  });
}

$('#bigPicModal').on('click', function(event) {
    if (event.target.id === 'popUpImage') {
        // Prevent modal from closing when image is clicked
        event.stopPropagation();
    } else {
        // Close modal when outside the image is clicked
        $(this).modal('hide');
    }
});

  document.getElementById('popUpImage').addEventListener('touchstart', function(e) {
    isZooming = false;
    if (e.touches.length === 1) {
        // Single touch (possible swipe)
      const touch = e.touches[0];
      startX = touch.clientX;
      startY = touch.clientY;
      this.style.transition = 'none'; // Disable transition for immediate feedback
    } else if (e.touches.length === 2) {
        // Two fingers (possible pinch)
      isZooming = true;
      initialTouchDistance = getDistanceBetweenTouches(e.touches);
    }
  }, false);

  document.getElementById('popUpImage').addEventListener('touchmove', function(e) {
    if (!isZooming)
    {
      currentX = e.touches[0].clientX - startX;
      this.style.transform = `translateX(${currentX*.3}px)`; // Move image with touch
    }
    else if (isZooming) {
        // If there's more than one touch, it could be a pinch zoom
      e.preventDefault();
    }
  }, false);

  document.getElementById('popUpImage').addEventListener('touchend', function(e) {
    if (!isZooming) {
          this.style.transition = 'transform 0.3s ease'; // Re-enable transition

          const touch = e.changedTouches[0];
          const endX = touch.clientX;
          const endY = touch.clientY;

          const xDiff = Math.abs(startX - endX);
          const yDiff = Math.abs(startY - endY);

    if (xDiff > window.innerWidth * 0.15) { // Adjust the xDiff threshold as needed
        // Horizontal swipe detected
      changeImage(startX > endX ? 1 : -1);
    }else {
        this.style.transform = ''; // Return image to original position
      }
    }
  }, false);
